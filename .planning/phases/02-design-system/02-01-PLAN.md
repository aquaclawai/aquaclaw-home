The updated layout.tsx should be:

```tsx
// src/app/layout.tsx
import type { Metadata } from 'next'
import { Fredoka, Nunito } from 'next/font/google'
import { GeistMono } from 'geist/font/mono'
import './globals.css'

const fredoka = Fredoka({
  subsets: ['latin'],
  variable: '--font-fredoka',
  display: 'swap',
  weight: ['400', '500', '600', '700'],
})

const nunito = Nunito({
  subsets: ['latin'],
  variable: '--font-nunito',
  display: 'swap',
})

export const metadata: Metadata = {
  metadataBase: new URL(
    process.env.VERCEL_URL
      ? `https://${process.env.VERCEL_URL}`
      : 'http://localhost:3000'
  ),
  title: {
    template: '%s | AquaClaw.ai',
    default: 'AquaClaw.ai — An AI-Operated Website',
  },
  openGraph: {
    siteName: 'AquaClaw.ai',
    type: 'website',
  },
}

export default function RootLayout({
  children,
}: {
  children: React.ReactNode
}) {
  return (
    <html lang="en" className={`${fredoka.variable} ${nunito.variable} ${GeistMono.variable}`}>
      <body>{children}</body>
    </html>
  )
}
```

Key points:
- Fredoka loads weights 400, 500, 600, 700 (variable font supports 300-700 but we specify the weights we need)
- Nunito is a variable-weight font, no weight restriction needed
- GeistMono is imported from the already-installed `geist` package
- All three set CSS variables that `@theme inline` in globals.css references
- `display: 'swap'` prevents FOIT (flash of invisible text)
- Preserve existing metadata configuration exactly as-is
