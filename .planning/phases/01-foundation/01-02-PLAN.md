Step 1 — Create src/app/[lang]/layout.tsx. This wraps every locale-prefixed page. It must NOT re-declare html/body tags (root layout in src/app/layout.tsx owns those). It provides the locale context for child pages:
```typescript
// src/app/[lang]/layout.tsx
import { ReactNode } from 'react'

interface LangLayoutProps {
  children: ReactNode
  params: Promise<{ lang: string }>
}

export default async function LangLayout({ children, params }: LangLayoutProps) {
  const { lang } = await params
  return (
    <div lang={lang}>
      {children}
    </div>
  )
}
```
NOTE: In Next.js 16, `params` in layouts and pages is a Promise — must `await params` before destructuring. This is a Next.js 16 async params requirement.

Step 2 — Create src/app/[lang]/page.tsx as a placeholder homepage. Uses getDictionary (never hardcodes strings per D-14):
```typescript
// src/app/[lang]/page.tsx
import { getDictionary, type Locale } from '@/lib/i18n/getDictionary'

interface HomePageProps {
  params: Promise<{ lang: string }>
}

export default async function HomePage({ params }: HomePageProps) {
  const { lang } = await params
  const dict = await getDictionary(lang as Locale)

  return (
    <main>
      <h1>{dict.site.name}</h1>
      <p>{dict.site.tagline}</p>
    </main>
  )
}
```
